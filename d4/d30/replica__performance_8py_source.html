32 """
33 Calculate latency and throughput metrics from log data.

35 metrics = {}

38 raw_commit_ns = data.get('commit_time', 0)
39 raw_exec_ns = data.get('execution_time', 0)
40
41 metrics['raw_propose_time'] = raw_prop_ns
42 metrics['raw_prepare_time'] = raw_prepare_ns
43 metrics['raw_commit_time'] = raw_commit_ns
44 metrics['raw_execution_time'] = raw_exec_ns
45
46 # convert nano -> ms:
47 execution_time_ms = (int(raw_exec_ns) / 1e6) if raw_exec_ns > 0 else None
48 propose_pre_prepare_time_ms = (int(raw_prop_ns) / 1e6) if raw_prop_ns > 0 else None
49 commit_time_ms = (int(raw_commit_ns) / 1e6) if raw_commit_ns > 0 else None
50 prepare_time_ms = (int(raw_prepare_ns) / 1e6) if raw_prepare_ns > 0 else None
51
52
53 if execution_time_ms is not None and propose_pre_prepare_time_ms is not None:
54 metrics['end_to_end_latency_ms'] = execution_time_ms - propose_pre_prepare_time_ms
55 else:
56 metrics['end_to_end_latency_ms'] = None
57
58 if raw_exec_ns > 0 and raw_commit_ns > 0:
59 metrics['execution_latency_ms'] = (int(raw_exec_ns) - int(raw_commit_ns))/1e6
60 else:
61 metrics['execution_latency_ms'] = None
62
63 if raw_prepare_ns > 0 and raw_prop_ns > 0:
64 metrics['consensus_latency_ms'] = (int(raw_prepare_ns) - int(raw_prop_ns))/1e6
65 else:
66 metrics['consensus_latency_ms'] = None
67
68 if raw_commit_ns > 0 and raw_prepare_ns > 0:
69 metrics['commit_latency_ms'] = (int(raw_commit_ns) - int(raw_prepare_ns))/1e6
70 else:
71 metrics['commit_latency_ms'] = None
72
73 # Store additional info
74 metrics['txn_number'] = data.get('txn_number',0)
75 metrics['replica_id'] = data.get('replica_id', 0)
76 metrics['primary_id'] = data.get('primary_id', 0)

131 def calculate_summary_stats(self, metrics_list: List[Dict[str, Any]]) -> Dict[str, Any]:
132 """
133 Calculate summary statistics from all metrics.
134 """
135 if not metrics_list:
136 return {}
137
138 # Filter out extreme latency values
139 def filter_valid_latencies(values):
140 return [v for v in values if v is not None and 0 <= v <= 60000]
141
142 # Extract numeric metrics
143 latency_fields = [
144 'end_to_end_latency_ms',
145 'execution_latency_ms',
146 'consensus_latency_ms',
147 'commit_latency_ms'
148 ]
149

151 'total_entries': len(metrics_list),
152 'averages': {},
153 'min_values': {},
154 'max_values': {},
155 'valid_counts': {}

157
158 # Process latency fields with filtering
159 for field in latency_fields:
160 all_values = [m[field] for m in metrics_list if field in m and m[field] is not None]
161 valid_values = filter_valid_latencies(all_values)
162
163 summary['valid_counts'][field] = len(valid_values)
164
165 if valid_values:
166 summary['averages'][field] = statistics.mean(valid_values)
167 summary['min_values'][field] = min(valid_values)
168 summary['max_values'][field] = max(valid_values)
169
170 # Report filtered values
171 if len(all_values) != len(valid_values):
172 print(f"Warning: {field} had {len(all_values) - len(valid_values)} invalid values filtered out")
173
174 # Calculate throughput
175 self.calculate_throughput(metrics_list, summary)

179 def calculate_throughput(self, metrics_list: List[Dict[str, Any]], summary: Dict[str, Any]):
180 """
181 Calculate throughput metrics.
182 """
183 if not metrics_list:
184 return
185
186 #First transaction seen
187 pre_propose_times = [m['raw_propose_time'] for m in metrics_list if m.get('raw_propose_time', 0) > 0]
188 first_propose = min(pre_propose_times)
189
190 #Last transaction committed
191 execution_times = [m['raw_execution_time'] for m in metrics_list if m.get('raw_execution_time', 0) > 0]
192 last_execute =max(execution_times)
193
194 # Get time range
195 time_range = (last_execute - first_propose)
196 time_range_s = time_range/1e9
197
198 #Get the total transactions
199 total_transactions = len(metrics_list)
200
201 if time_range_s > 0:
202 summary['throughput_tps'] = total_transactions / time_range_s
203 summary['time_range_seconds'] = time_range_s
204 summary['total_transactions'] = total_transactions
205
206 # Per-replica throughput
207 replica_data = {}
208 for i in range(16):
209 replica_prepropose = []
210 replica_execution = []
211 transaction_count = 0
212
213 for m in metrics_list:
214 rep_id = m.get('replica_id')
215 if rep_id == i:
216 raw_prop = m.get("raw_propose_time", 0)
217 raw_exec = m.get("raw_execution_time", 0)
218 if raw_prop > 0 and raw_exec > 0:
219 replica_prepropose.append(raw_prop)
220 replica_execution.append(raw_exec)
221 transaction_count += 1
222
223 if replica_prepropose and replica_execution:
224 replica_first_time = min(replica_prepropose)
225 replica_last_time = max(replica_execution)
226 replica_timerange_s = (replica_last_time - replica_first_time) / 1e9
227 if replica_timerange_s > 0:
228 replica_tps = transaction_count / replica_timerange_s
229 else:
230 replica_tps = 0.0
231 else:
232 replica_tps = 0.0
233
234 replica_data[f"replica_{i}_tps"] = replica_tps
235
236
237 #Store in summary
238 summary["per_replica_tps"] = replica_data
